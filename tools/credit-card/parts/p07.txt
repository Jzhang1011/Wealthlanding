function generateStickers() {
  if (wallet.size === 0) { alert("Please add cards to your wallet first to generate stickers!"); return; }

  const list = document.getElementById('sticker-list');
  list.innerHTML = '';
  const cardWins = {};
  
  CATEGORIES.forEach(cat => {
    const owned = [...cat.entries].filter(e => wallet.has(e.card)).sort((a, b) => effVal(b) - effVal(a));
    if (owned.length > 0) {
      const cardId = owned[0].card;
      if (!cardWins[cardId]) cardWins[cardId] = { card: CARDS[cardId], wins: [] };
      cardWins[cardId].wins.push({ icon: cat.icon, name: cat.name, rate: owned[0].rate });
    }
  });

  Object.values(cardWins).forEach(group => {
    const lines = group.wins.map(w => {
      const simpleName = w.name.split(' (')[0];
      return `<span class="whitespace-nowrap font-medium">${w.icon} ${simpleName} ${w.rate}</span>`;
    }).join('<span class="mx-1 text-slate-300">|</span>');

    list.innerHTML += `
      <div style="width: 2.25in; min-height: 0.5in;" class="border border-dashed border-slate-400 p-1 bg-white text-[8px] leading-tight">
        <div class="font-extrabold text-[9px] mb-0.5 border-b border-slate-200 pb-0.5 uppercase tracking-tight text-slate-800">
          ${group.card.name}
        </div>
        <div class="flex flex-wrap text-slate-700 leading-[1.3] mt-0.5">
          ${lines}
        </div>
      </div>
    `;
  });

  document.body.classList.remove('print-reference'); document.body.classList.add('print-stickers');
  setTimeout(() => { window.print(); }, 100);
}

function generateReferenceCard() {
  if (wallet.size === 0) { alert("Please add cards to your wallet first!"); return; }

  const list = document.getElementById('reference-list');
  list.innerHTML = '';
  
  CATEGORIES.forEach(cat => {
    const owned = [...cat.entries].filter(e => wallet.has(e.card)).sort((a, b) => effVal(b) - effVal(a));
      
    if (owned.length > 0) {
      const bestCard = CARDS[owned[0].card];
      const simpleName = cat.name.split(' (')[0];
      
      list.innerHTML += `
        <div class="flex justify-between border-b border-slate-200 pb-[1px]">
          <span class="font-bold pr-1 truncate">${cat.icon} ${simpleName}</span>
          <span class="text-right whitespace-nowrap text-slate-700">${bestCard.name} <span class="font-bold text-black ml-0.5">${owned[0].rate}</span></span>
        </div>
      `;
    }
  });

  document.body.classList.remove('print-stickers'); document.body.classList.add('print-reference');
  setTimeout(() => { window.print(); }, 100);
}
